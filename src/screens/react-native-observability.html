Sentry Setup

Sentry Initialization & React Navigation Instrumentation

import * as Sentry from '@sentry/react-native';

Sentry.init({
  dsn: 'https://key@sentry.io/project-id',
  tracesSampleRate: 0.2,
  enableNative: true,
  release: 'com.myapp@1.4.0+42',
  beforeSend(event) {
    // Strip sensitive tokens from URL breadcrumbs
    if (event.breadcrumbs) {
      event.breadcrumbs = event.breadcrumbs.filter(
        b => !b.data?.url?.includes('/auth/')
      );
    }
    return event;
  },
});

export default Sentry.wrap(App);
