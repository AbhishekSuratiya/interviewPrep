Detox E2E Example

Checkout Flow Test in Detox

describe('Checkout Flow', () => {
  beforeAll(async () => {
    await device.launchApp({ newInstance: true });
  });

  beforeEach(async () => {
    await device.reloadReactNative();
  });

  it('should complete checkout successfully', async () => {
    await element(by.id('cart-button')).tap();
    await element(by.id('checkout-button')).tap();
    await element(by.id('input-card')).typeText('4242424242424242');
    await element(by.text('Confirm Payment')).tap();

    // Detox auto-waits for JS thread to settle before checking visibility
    await expect(element(by.text('Order Successful'))).toBeVisible();
  });
});
