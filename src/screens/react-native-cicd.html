Configuration Code

Fastlane Fastfile & EAS Build Config

# Fastfile lane
lane :beta do
  increment_build_number
  match(type: 'appstore') # Sync certs from git repo
  gym(scheme: 'MyApp', configuration: 'Release')
  pilot(skip_waiting_for_build_processing: true) # Upload to TestFlight
end

# eas.json
{
  "build": {
    "production": {
      "autoIncrement": true,
      "channel": "production"
    }
  }
}
