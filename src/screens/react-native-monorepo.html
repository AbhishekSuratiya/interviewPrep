Metro Bundler Config

watchFolders & Resolvers

Metro must be explicitly configured with watchFolders pointing to root monorepo packages/ to bundle symlinked packages correctly.

Metro Monorepo Config

metro.config.js for Symlinked Monorepos

const { getDefaultConfig } = require('@react-native/metro-config');
const path = require('path');

const projectRoot = __dirname;
const workspaceRoot = path.resolve(projectRoot, '../..');

const config = getDefaultConfig(projectRoot);

// Watch all packages in workspace root
config.watchFolders = [
  path.resolve(workspaceRoot, 'packages/ui-tokens'),
  path.resolve(workspaceRoot, 'packages/api-client'),
];

// Force Metro to resolve modules from project node_modules
config.resolver.nodeModulesPaths = [
  path.resolve(projectRoot, 'node_modules'),
  path.resolve(workspaceRoot, 'node_modules'),
];

module.exports = config;
