import NetInfo from '@react-native-community/netinfo';

// Queue mutation locally
async function queueOfflineMutation(type: string, payload: any) {
  await db.execute('INSERT INTO outbox (type, payload, timestamp) VALUES (?, ?, ?)', [
    type,
    JSON.stringify(payload),
    Date.now(),
  ]);

  // Flush immediately if network is reachable
  const state = await NetInfo.fetch();
  if (state.isConnected && state.isInternetReachable) {
    await flushOutboxQueue();
  }
}
