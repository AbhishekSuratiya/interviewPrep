Code Snippet

InteractionManager & Deferral Pattern

import { InteractionManager, useEffect } from 'react';

function DashboardScreen() {
  useEffect(() => {
    // Wait for screen transition animation to finish
    const task = InteractionManager.runAfterInteractions(() => {
      // Execute expensive data calculation / fetch
      fetchAnalyticsData();
    });

    return () => task.cancel();
  }, []);

  return <View>{/* UI */}</View>;
}
